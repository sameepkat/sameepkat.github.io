Syntax Showcase 

use io
use array
use math
use time
use terminal
use string

fn banner(title) {
    println("==============================")
    println(title)
    println("==============================")
}

fn factorial(n) {
    if n <= 1 {
        return 1
    }
    return n * factorial(n - 1)
}

fn fib(n) {
    if n <= 1 {
        return n
    }
    return fib(n - 1) + fib(n - 2)
}

fn is_prime(n) {
    if n < 2 {
        return false
    }

    d = 2
    while d * d <= n {
        if n % d == 0 {
            return false
        }
        d = d + 1
    }

    return true
}

fn print_array(items) {
    print("[")
    i = 0
    while i < len(items) {
        print(items[i])
        if i < len(items) - 1 {
            print(", ")
        }
        i = i + 1
    }
    println("]")
}

fn sum_array(items) {
    total = 0
    i = 0
    while i < len(items) {
        total = total + items[i]
        i = i + 1
    }
    return total
}

fn max_array(items) {
    best = items[0]
    i = 1
    while i < len(items) {
        if items[i] > best {
            best = items[i]
        }
        i = i + 1
    }
    return best
}

fn reverse_array(items) {
    result = []
    i = len(items) - 1
    while i >= 0 {
        push(result, items[i])
        i = i - 1
    }
    return result
}

fn filter_odd_numbers(items) {
    result = []
    i = 0
    while i < len(items) {
        value = items[i]

        if value % 2 == 0 {
            i = i + 1
            continue
        }

        push(result, value)
        i = i + 1
    }
    return result
}

fn find_first_prime(items) {
    i = 0
    while i < len(items) {
        if is_prime(items[i]) {
            return items[i]
        }
        i = i + 1
    }
    return nil
}

fn count_until_limit(limit) {
    i = 0
    while true {
        if i > limit {
            break
        }
        println(i)
        i = i + 1
    }
}

fn draw_progress(label, steps) {
    print(label)
    print(" ")

    i = 0
    while i < steps {
        print("#")
        flush()
        sleep_ms(20)
        i = i + 1
    }

    println(" done")
}

fn make_student(name, age, scores) {
    return {
        name: name,
        age: age,
        scores: scores,
        active: true,
        note: nil
    }
}

fn average_score(student) {
    scores = student.scores
    if len(scores) == 0 {
        return 0
    }
    return sum_array(scores) / len(scores)
}

fn print_student(student) {
    println("Student record:")
    print("name: ")
    println(student.name)
    print("age: ")
    println(student.age)
    print("scores: ")
    print_array(student.scores)
    print("average: ")
    println(average_score(student))
}

fn math_demo() {
    banner("math demo")

    angle = 1.0
    print("sin(1): ")
    println(sin(angle))
    print("cos(1): ")
    println(cos(angle))
    print("tan(1): ")
    println(tan(angle))
    print("floor(3.9): ")
    println(floor(3.9))
    print("ceil(3.1): ")
    println(ceil(3.1))
}

fn array_demo() {
    banner("array demo")

    numbers = [1, 2, 3, 4, 5, 13, 21]

    print("numbers: ")
    print_array(numbers)

    print("sum: ")
    println(sum_array(numbers))

    print("max: ")
    println(max_array(numbers))

    print("reversed: ")
    print_array(reverse_array(numbers))

    print("odds: ")
    print_array(filter_odd_numbers(numbers))

    print("first prime: ")
    println(find_first_prime(numbers))
}

fn recursion_demo() {
    banner("recursion demo")

    i = 0
    while i <= 7 {
        print("factorial(")
        print(i)
        print(") = ")
        println(factorial(i))
        i = i + 1
    }

    println("")

    i = 0
    while i <= 8 {
        print("fib(")
        print(i)
        print(") = ")
        println(fib(i))
        i = i + 1
    }
}

fn object_demo() {
    banner("object demo")

    sameep = make_student("Sameep", 21, [90, 87, 94, 99])
    sameep.note = "likes compilers and tiny models"

    print_student(sameep)

    if sameep.active and average_score(sameep) > 90 {
        println("status: excellent")
    } else {
        println("status: keep practicing")
    }
}

fn loop_demo() {
    banner("loop demo")

    println("count until 3 using break:")
    count_until_limit(3)

    println("skip even numbers using continue:")
    i = 0
    while i < 10 {
        if i % 2 == 0 {
            i = i + 1
            continue
        }

        println(i)
        i = i + 1
    }
}

fn string_demo() {
   banner("string conversion demo")
   a = number("123")
   b = number("1.34")

   print("number(123) + 7 = ")
   println(a + 7)

   print("number(1.34) + 1 = ")
   println(b + 1)

   print("string(123) = ")
   println(string(123))

   print("string(true) = ")
   println(string(123))

   print("string(true) = ")
   println(string(true))

   print("string(nil) = ")
   println(string(nil))
}

fn main() {
    clear()
    hide_cursor()

    banner("Lum syntax showcase")

    recursion_demo()
    array_demo()
    object_demo()
    loop_demo()
    math_demo()
    string_demo()

    show_cursor()


    println("All done!")
}

main()
